SELECT SGL_ORGAO
FROM DDA.ORGAO
WHERE
    SITUACAO = 'ATIVO'
    AND (
        SGL_ORGAO_HIERARQUIA LIKE '%ASCOM%'
        OR SGL_ORGAO_HIERARQUIA LIKE '%CEPRES%'
        OR (SGL_ORGAO_HIERARQUIA LIKE '%SGEST%' AND SITUACAO = 'ATIVO')
        OR SGL_ORGAO_HIERARQUIA LIKE '%NUDIV%'
        OR SGL_ORGAO_HIERARQUIA LIKE '%ESINT%'
        OR SGL_ORGAO = 'PRSECR'
        OR (
                SGL_ORGAO_HIERARQUIA LIKE '%SRPSF%'
                OR SGL_ORGAO_HIERARQUIA LIKE '%COASAS%'
                OR SGL_ORGAO_HIERARQUIA LIKE '%GBDGER%'
                OR SGL_ORGAO_HIERARQUIA LIKE '%ASQUALOG%'
		OR SGL_ORGAO_HIERARQUIA LIKE '%ADVOSF%'
		OR SGL_ORGAO_HIERARQUIA LIKE '%NGMUS%'
            )
    )
ORDER BY
    CASE
        WHEN SGL_ORGAO_HIERARQUIA LIKE '%ASCOM%'    THEN 1
        WHEN SGL_ORGAO_HIERARQUIA LIKE '%CEPRES%'   THEN 2
        WHEN SGL_ORGAO_HIERARQUIA LIKE '%SGEST%'    THEN 3
        WHEN SGL_ORGAO_HIERARQUIA LIKE '%NUDIV%'   THEN 4
        WHEN SGL_ORGAO_HIERARQUIA LIKE '%ESINT%'    THEN 5
        WHEN SGL_ORGAO = 'PRSECR'                   THEN 6
        WHEN SGL_ORGAO_HIERARQUIA LIKE '%SRPSF%'    THEN 7
        WHEN SGL_ORGAO_HIERARQUIA LIKE '%COASAS%'   THEN 8
        WHEN SGL_ORGAO_HIERARQUIA LIKE '%GBDGER%'   THEN 9
	WHEN SGL_ORGAO_HIERARQUIA LIKE '%ADVOSF%'   THEN 10
        WHEN SGL_ORGAO_HIERARQUIA LIKE '%ASQUALOG%' THEN 11
	WHEN SGL_ORGAO_HIERARQUIA LIKE '%NGMUS%'    THEN 12
    END,
    SGL_ORGAO_HIERARQUIA;
